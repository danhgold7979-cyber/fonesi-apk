cmake_minimum_required(VERSION 3.22)
project(FonesiRuntime LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} -x assembler-with-cpp")
if(NOT EXISTS "${DOBBY_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/prepare_source_engine.py first")
endif()
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(Plugin.ImportTableReplace OFF CACHE BOOL "" FORCE)
set(Plugin.Android.BionicLinkerUtil OFF CACHE BOOL "" FORCE)
add_subdirectory("${DOBBY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/dobby")

set(FONESI_SOURCES PathRules.cpp NativeHooks.cpp)
if(ANDROID)
    add_library(fonesi_runtime SHARED ${FONESI_SOURCES} NativeBridge.cpp)
    target_compile_definitions(fonesi_runtime PRIVATE _GNU_SOURCE)
    target_compile_options(fonesi_runtime PRIVATE -Wall -Wextra -fvisibility=hidden -fstack-protector-strong)
    target_link_options(fonesi_runtime PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,relro -Wl,-z,now)
    target_link_options(fonesi_runtime PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    target_link_libraries(fonesi_runtime PRIVATE dobby_static log dl)
else()
    add_executable(fonesi_native_test ${FONESI_SOURCES} ../../../tests/NativeRuntimeTest.cpp)
    target_include_directories(fonesi_native_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(fonesi_native_test PRIVATE _GNU_SOURCE)
    target_compile_options(fonesi_native_test PRIVATE -Wall -Wextra)
    target_link_libraries(fonesi_native_test PRIVATE dobby_static dl pthread)
endif()
